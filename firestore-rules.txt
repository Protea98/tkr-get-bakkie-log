rules_version = '2';
// TKR Operations app – rules updated 21 Nov 2026 (GET Forecast open to view; members sign in with name and PIN)
service cloud.firestore {
  match /databases/{database}/documents {

    // ---------- who is who ----------
    function signedIn() { return request.auth != null; }
    function myEmail() { return signedIn() ? request.auth.token.get('email', '').lower() : ''; }
    function isAdmin() { return myEmail() in ['muneer@tkroperations.com', 'amogelang@tkroperations.com']; }
    // supervisor (Edwin): full rights in Stores and the GET Bakkie Log only
    function isSupervisor() {
      return signedIn() && exists(/databases/$(database)/documents/config/roles)
        && myEmail() in get(/databases/$(database)/documents/config/roles).data.get('supervisors', []);
    }
    function sectionBoss() { return isAdmin() || isSupervisor(); }
    function isManager() {
      return isAdmin() || (exists(/databases/$(database)/documents/config/roles)
        && myEmail() in get(/databases/$(database)/documents/config/roles).data.get('managers', []));
    }

    // ---------- Daily Task Board ----------
    function ownsPerson(personId) {
      return signedIn() && personId is string
        && exists(/databases/$(database)/documents/team/$(personId))
        && get(/databases/$(database)/documents/team/$(personId)).data.get('uid', '') == request.auth.uid;
    }

    match /config/{docId} {
      allow read: if signedIn() && !(docId in ['attEmployees', 'attSettings']);
      allow write: if isAdmin() && !(docId in ['hrAccess', 'trainingAccess']);
    }
    match /team/{id} {
      allow read: if signedIn();
      allow create, delete: if isAdmin();
      allow update: if isAdmin() || (
        signedIn()
        && resource.data.get('uid', null) == null
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.authEmail == myEmail()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['uid', 'authEmail', 'pinSetAt'])
      );
    }
    match /tasks/{id} {
      allow read: if isManager() || ownsPerson(resource.data.person);
      allow create: if isAdmin() || ownsPerson(request.resource.data.person);
      allow update: if isAdmin() || (ownsPerson(resource.data.person) && request.resource.data.person == resource.data.person);
      allow delete: if isAdmin() || ownsPerson(resource.data.person);
    }

    // ---------- GET Bakkie Log ----------
    function approvedOp() {
      return signedIn() && exists(/databases/$(database)/documents/getOperators/$(request.auth.uid))
        && get(/databases/$(database)/documents/getOperators/$(request.auth.uid)).data.get('approved', false) == true;
    }
    function exportGrant() {
      return signedIn() && exists(/databases/$(database)/documents/getExportGrants/$(request.auth.uid))
        && get(/databases/$(database)/documents/getExportGrants/$(request.auth.uid)).data.get('until', 0) > request.time.toMillis();
    }

    match /getOperators/{uid} {
      allow read: if signedIn();
      allow create: if signedIn() && uid == request.auth.uid && request.resource.data.approved == false;
      allow update, delete: if sectionBoss();
    }
    match /getExportRequests/{id} {
      allow read: if sectionBoss() || (signedIn() && resource.data.opUid == request.auth.uid);
      allow create: if signedIn() && request.resource.data.opUid == request.auth.uid && request.resource.data.status == 'pending';
      allow update, delete: if sectionBoss();
    }
    match /getExportGrants/{uid} {
      allow read: if sectionBoss() || (signedIn() && uid == request.auth.uid);
      allow write: if sectionBoss();
    }
    match /getShifts/{id} {
      allow read: if sectionBoss() || exportGrant() || (signedIn() && resource.data.ownerUid == request.auth.uid);
      allow create: if approvedOp() && request.resource.data.ownerUid == request.auth.uid;
      allow update: if approvedOp() && resource.data.ownerUid == request.auth.uid && request.resource.data.ownerUid == request.auth.uid;
      allow delete: if sectionBoss() || (approvedOp() && (resource == null || resource.data.ownerUid == request.auth.uid));
    }
    match /getPhotos/{id} {
      allow read: if sectionBoss() || exportGrant() || (signedIn() && resource.data.ownerUid == request.auth.uid);
      allow create: if approvedOp() && request.resource.data.ownerUid == request.auth.uid;
      allow update: if approvedOp() && resource.data.ownerUid == request.auth.uid && request.resource.data.ownerUid == request.auth.uid;
      allow delete: if sectionBoss() || (approvedOp() && (resource == null || resource.data.ownerUid == request.auth.uid));
    }

    // ---------- GET Forecast ----------
    // ---------- Control Room (controllers capture on the control room PC; managers see it) ----------
    function ctrlUser() {
      return signedIn() && exists(/databases/$(database)/documents/config/ctrlAccess)
        && myEmail() in get(/databases/$(database)/documents/config/ctrlAccess).data.get('emails', []);
    }
    // only Control Room app v206 or newer may write (older copies still open on a PC can't change anything)
    match /ctrl/{id} { allow read: if signedIn(); allow create, update: if signedIn() && request.resource.data.get('v', 0) >= 206; allow delete: if signedIn(); }   // the control room PCs sign in by themselves (no email); controllers use their name and PIN in the page

    // GET Forecast: anyone who opens it can see the dashboard; admins change it
    match /fcMetres/{id}   { allow read: if signedIn(); allow write: if isAdmin(); }
    match /fcIssues/{id}   { allow read: if signedIn(); allow write: if isAdmin(); }
    match /fcCounts/{id}   { allow read: if signedIn(); allow write: if isAdmin(); }
    match /fcSettings/{id} { allow read: if signedIn(); allow write: if isAdmin(); }
    // GET Forecast members: register with a name and PIN, an admin approves, then they sign in and stay signed in
    match /fcMembers/{id} {
      allow read: if signedIn();
      allow create: if signedIn() && request.resource.data.status == 'pending';
      allow update, delete: if isAdmin();
    }
    match /fcMemberPins/{id} {
      allow read: if false;
      allow create: if signedIn() && !exists(/databases/$(database)/documents/fcMemberPins/$(id));
      allow update, delete: if isAdmin();
    }
    match /fcMemberSess/{uid} {
      allow read: if signedIn() && (request.auth.uid == uid || isAdmin());
      allow delete: if signedIn() && (request.auth.uid == uid || isAdmin());
      allow create: if signedIn() && request.auth.uid == uid
        && get(/databases/$(database)/documents/fcMembers/$(request.resource.data.member)).data.status == 'approved'
        && request.resource.data.pin == get(/databases/$(database)/documents/fcMemberPins/$(request.resource.data.member)).data.pin;
    }

    // ---------- Stores (Drill Stores, Rompad Stores, PPE) ----------
    function stStaff() {
      return signedIn() && exists(/databases/$(database)/documents/stUsers/$(request.auth.uid))
        && get(/databases/$(database)/documents/stUsers/$(request.auth.uid)).data.get('approved', false) == true;
    }
    function stReader() { return isManager() || isSupervisor() || stStaff(); }

    match /stUsers/{uid} {
      allow read: if signedIn();
      allow create: if signedIn() && uid == request.auth.uid && request.resource.data.approved == false;
      allow update, delete: if sectionBoss();
    }
    match /stMoves/{id} {
      allow read: if stReader() || (signedIn() && resource.data.store == 'drill');   // Drill Stores GET shows on the GET Forecast
      allow create: if (stStaff() || sectionBoss()) && request.resource.data.byUid == request.auth.uid
        && request.resource.data.kind in ['issue', 'receive'] && request.resource.data.store in ['drill', 'rompad'];
      allow update: if sectionBoss();
      allow delete: if sectionBoss() || (stStaff() && resource.data.byUid == request.auth.uid);
    }
    match /stCounts/{id} {
      allow read: if stReader() || (signedIn() && resource.data.store == 'drill');
      allow create: if (stStaff() || sectionBoss()) && request.resource.data.byUid == request.auth.uid;
      allow update, delete: if sectionBoss();
    }
    match /stReports/{id} {
      allow read: if stReader();
      allow create: if (stStaff() || sectionBoss()) && request.resource.data.byUid == request.auth.uid;
      allow update, delete: if sectionBoss();
    }
    match /stReportPhotos/{id} {
      allow read: if stReader();
      allow create: if (stStaff() || sectionBoss()) && request.resource.data.byUid == request.auth.uid;
      allow update, delete: if sectionBoss();
    }
    match /stConfig/status {
      allow read: if stReader();
      allow write: if stStaff() || sectionBoss();
    }
    match /stConfig/{docId} {
      allow read: if stReader() || (signedIn() && docId == 'items');
      allow write: if sectionBoss();
    }

    // ---------- Attendance ----------
    function attAdmin() { return myEmail() in ['muneer@tkroperations.com', 'amogelang@tkroperations.com', 'amogelangmabilu56@gmail.com']; }
    function attRoles() { return get(/databases/$(database)/documents/config/attendance).data; }
    function attHasRoles() { return exists(/databases/$(database)/documents/config/attendance); }
    function attCapturer() { return attAdmin() || (attHasRoles() && myEmail() in attRoles().get('capturers', [])); }
    function attViewer() { return attCapturer() || isManager() || (attHasRoles() && myEmail() in attRoles().get('viewers', [])); }

    match /config/attendance {
      allow read: if myEmail() != '';
      allow write: if attAdmin();
    }
    match /config/attEmployees {
      allow read: if attViewer();
      allow write: if attCapturer();
    }
    match /config/attSettings {
      allow read: if attViewer();
      allow write: if attCapturer();
    }
    match /attCycles/{id} {
      allow read: if attViewer();
      allow write: if attAdmin();
    }
    match /attMarks/{id} {
      allow read: if attViewer();
      allow write: if attCapturer();
    }
    match /attRota/{id} {
      allow read: if attViewer();
      allow write: if attCapturer();
    }
    match /attMonths/{month}/sheets/{sheet} {
      allow read: if attViewer();
      allow write: if attCapturer();
    }
    // ---------- HR (access given by Amogelang only) ----------
    function hrAdmin() { return myEmail() in ['amogelang@tkroperations.com', 'amogelangmabilu56@gmail.com']; }
    function hrHas() { return exists(/databases/$(database)/documents/config/hrAccess); }
    function hrAcc() { return get(/databases/$(database)/documents/config/hrAccess).data; }
    function hrEditor() { return hrAdmin() || (hrHas() && myEmail() in hrAcc().get('editors', [])); }
    function hrViewer() { return hrEditor() || (hrHas() && myEmail() in hrAcc().get('viewers', [])); }
    match /config/hrAccess {
      allow read: if signedIn();
      allow write: if hrAdmin();
    }
    match /hr/{section}/items/{id} {
      allow read: if hrViewer();
      allow write: if hrEditor();
    }
    // ---------- Training Matrix (access given by Muneer and Amogelang) ----------
    function tmHas() { return exists(/databases/$(database)/documents/config/trainingAccess); }
    function tmAcc() { return get(/databases/$(database)/documents/config/trainingAccess).data; }
    function tmEditor() { return isAdmin() || hrAdmin() || (tmHas() && myEmail() in tmAcc().get('editors', [])); }
    function tmViewer() { return tmEditor() || (tmHas() && myEmail() in tmAcc().get('viewers', [])); }
    match /config/trainingAccess {
      allow read: if signedIn();
      allow write: if isAdmin() || hrAdmin();
    }
    match /training/{id} {
      allow read: if tmViewer();
      allow write: if tmEditor();
    }
  }
}
